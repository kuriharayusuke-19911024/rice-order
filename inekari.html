} } function respond(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } /* ================= 読み込み ================= */ function loadAll() { const ss = SpreadsheetApp.getActiveSpreadsheet();
